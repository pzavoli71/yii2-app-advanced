CREATE TABLE `session` (
  `id` char(40) COLLATE utf8mb4_general_ci NOT NULL,
  `expire` int DEFAULT NULL,
  `data` longblob,
  `user_id` int DEFAULT NULL,
  `browser_platform` varchar(200) COLLATE utf8mb4_general_ci DEFAULT NULL,
  `last_write` datetime DEFAULT CURRENT_TIMESTAMP,
  `ipaddress` varchar(30) COLLATE utf8mb4_general_ci DEFAULT NULL,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
